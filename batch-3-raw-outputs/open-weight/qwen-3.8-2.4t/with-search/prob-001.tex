\begin{thm}\label{thm:main}
Let $P$ be a monoid in the sense of the question, let $A_P$ be the category of homomorphisms $P\to Q$, and let $B_P\subset A_P$ be the full subcategory of maps $P\to P[-x]^\sharp$.  Equip $A_P^{\rm op}$ with the birhizomic topology and $B_P^{\rm op}$ with the trivial topology.  Let
\[
\pi^{-1}:\Sh(B_P^{\rm op})\longrightarrow \Sh(A_P^{\rm op})
\]
be the functor obtained by left Kan extension followed by sheafification, and let $\pi_*$ be its right adjoint.  Then for every presheaf of abelian groups $F$ on $B_P^{\rm op}$ the canonical morphism
\[
F\longrightarrow R\pi_*\pi^{-1}F
\]
is a quasi-isomorphism.
\end{thm}

The proof is organised as follows.  In \S\ref{sec:formal} we isolate the formal consequences of two facts: exactness of $\pi^{-1}$ and preservation of injectives.  In \S\ref{sec:filtered} we prove exactness of the left Kan extension step.  In \S\ref{sec:contract} we prove, using the defining lifting property of birhizomic morphisms, the local contractibility statement which implies that the underived unit is an isomorphism.  Finally, in \S\ref{sec:derived} we pass to derived categories.

\section{Notation and formal reductions}\label{sec:formal}

Write
\[
C:=A_P^{\rm op},\qquad D:=B_P^{\rm op},
\]
and let $i:D\hookrightarrow C$ be the inclusion.  Since $D$ carries the trivial topology, every presheaf on $D$ is a sheaf; hence
\[
\Sh(D)=\PSh(D).
\]

For $x\in P$ we write
\[
P_x:=P[-x]^\sharp
\]
for the initial sharp saturated quotient/localisation of $P$ in which $x$ becomes $0$.  Thus the objects of $B_P$ are the maps $P\to P_x$.

The functor $\pi^{-1}$ is by definition
\[
\pi^{-1}=a_C\circ \Lan_{i^{\rm op}},
\]
where $a_C$ is sheafification on $C$.  Its right adjoint is the restriction functor
\[
\pi_*:\Sh(C)\longrightarrow \Sh(D),\qquad G\longmapsto G|_D.
\]
Indeed, for $F\in \PSh(D)$ and $G\in \Sh(C)$ one has natural bijections
\[
\Hom_{\Sh(C)}(a_C\Lan_{i^{\rm op}}F,G)
\simeq
\Hom_{\PSh(C)}(\Lan_{i^{\rm op}}F,G)
\simeq
\Hom_{\PSh(D)}(F,G|_D).
\]

The strategy is:
\begin{enumerate}[label=(\alph*)]
\item prove that $\pi^{-1}$ is exact;
\item prove that $\pi^{-1}$ sends injective objects of $\PSh(D)$ to injective objects of $\Sh(C)$;
\item prove that the underived unit
\[
F\longrightarrow \pi_*\pi^{-1}F
\]
is an isomorphism for every presheaf $F$ on $D$.
\end{enumerate}
Then the derived statement follows formally by resolving $F$ by injectives.

\section{Exactness of $\pi^{-1}$}\label{sec:filtered}

We first prove that the left Kan extension part is exact.

For $Q\in A_P$, let $B_P/Q$ denote the comma category whose objects are morphisms $P_x\to Q$ in $A_P$ and whose morphisms are commutative triangles over $P$.  The value of the left Kan extension of $F$ at $Q$ is
\[
(\Lan F)(Q)=\varinjlim_{P_x\to Q}F(P_x).
\]

\begin{lem}\label{lem:filtered}
For every $Q\in A_P$, the category $B_P/Q$ is filtered.
\end{lem}
\begin{proof}
Let $q:P\to Q$ be the structure map.  A morphism $P_x\to Q$ exists exactly when $q(x)=0$.  Let
\[
K_q:=\{x\in P\mid q(x)=0\}.
\]
Because $Q$ is sharp and integral, $K_q$ is a face of $P$.  The category $B_P/Q$ is equivalent to the category of principal faces contained in $K_q$: an object is $x\in K_q$, and there is a morphism $x\to y$ precisely when the image of $x$ is $0$ in $P_y$, i.e.\ when $x$ lies in the face generated by $y$.

This category is nonempty because $0\in K_q$.  Given $x,y\in K_q$, the element $x+y$ lies in $K_q$, and both $x$ and $y$ lie in the face generated by $x+y$.  Hence there are morphisms
\[
x\longrightarrow x+y,\qquad y\longrightarrow x+y.
\]
Parallel morphisms are equal because maps between the localisations $P_x$ are unique when they exist.  Thus $B_P/Q$ is filtered.
\end{proof}

\begin{cor}\label{cor:Lan-exact}
The functor
\[
\Lan:\PSh(D)\longrightarrow \PSh(C)
\]
is exact.
\end{cor}
\begin{proof}
It is computed objectwise by filtered colimits of abelian groups, and filtered colimits are exact in $\mathbf{Ab}$.
\end{proof}

Sheafification of abelian presheaves on a Grothendieck site is exact.  Therefore:

\begin{cor}\label{cor:pi-inv-exact}
The functor
\[
\pi^{-1}=a_C\circ \Lan:\Sh(D)\to \Sh(C)
\]
is exact.
\end{cor}

\begin{lem}\label{lem:preserves-injectives}
The functor $\pi^{-1}$ preserves injective objects.
\end{lem}
\begin{proof}
Let $I$ be injective in $\PSh(D)$.  We must show that $\pi^{-1}I$ is injective in $\Sh(C)$.  Let $G'\hookrightarrow G$ be a monomorphism in $\Sh(C)$.  Since monomorphisms of sheaves of abelian groups are detected objectwise, the induced map
\[
\pi_*G'=G'|_D\longrightarrow G|_D=\pi_*G
\]
is a monomorphism of presheaves on $D$.  By adjunction,
\[
\Hom_{\Sh(C)}(G,\pi^{-1}I)
\simeq
\Hom_{\PSh(D)}(\pi_*G,I),
\]
and similarly for $G'$.  Since $I$ is injective in $\PSh(D)$, the restriction map
\[
\Hom_{\PSh(D)}(\pi_*G,I)
\longrightarrow
\Hom_{\PSh(D)}(\pi_*G',I)
\]
is surjective.  Hence
\[
\Hom_{\Sh(C)}(G,\pi^{-1}I)
\longrightarrow
\Hom_{\Sh(C)}(G',\pi^{-1}I)
\]
is surjective.  Thus $\pi^{-1}I$ is injective.
\end{proof}

\section{Local contractibility at objects of $B_P$}\label{sec:contract}

The essential geometric input is that birhizomic covers are contractible when tested on objects coming from $B_P$.

We first need faithful valuative points.

\begin{lem}\label{lem:faithful-valuative}
Let $M$ be an integral, saturated, sharp commutative monoid.  There exists a valuative monoid $V$ and an injective monoid homomorphism
\[
M\hookrightarrow V.
\]
\end{lem}
\begin{proof}
Since $M$ is saturated and sharp, $M^{\rm gp}$ is torsion-free: if $ng=0$ with $n>0$, then saturation gives $g\in M$ and $-g\in M$, hence $g=0$ by sharpness.  A torsion-free abelian group admits a total translation-invariant order extending any given proper cone.  Applying this to the cone $M\subset M^{\rm gp}$ gives a total order on $M^{\rm gp}$ extending the partial order defined by $M$.  Let $V$ be the positive cone of this total order.  Then $V$ is integral, saturated, sharp, and valuative, and the inclusion $M\subset V$ is injective.  See also the discussion of total preorders on monoids in~\href{https://ar5iv.labs.arxiv.org/html/2607.07473}{ar5iv.labs.arxiv.org}.
\end{proof}

For $x\in P$ write $b=P_x$.  If $q:P\to Q$ is an object of $A_P$ under $P$, let
\[
K_q:=\{a\in P\mid a\text{ maps to }0\text{ in }Q\}.
\]
If $q$ factors through $P_x$, then the face $F_x$ generated by $x$ is contained in $K_q$.

The following is the key contractibility statement.  It is the precise form of the valuative descent property encoded by the definition of birhizomic covers; the proof is the standard argument using the infinitesimal lifting property (see~\href{https://pub.math.leidenuniv.nl/~holmesdst/notes_from_talks/valuativisations_talk.pdf}{pub.math.leidenuniv.nl} for the valuativisation argument, and~\href{https://arxiv.org/pdf/2411.13662}{arxiv.org} for the finite presentation and Kummer-type stability properties of the relevant monoid maps).

\begin{lem}[Birhizomic contractibility]\label{lem:contract}
Let $b=P_x\in B_P$, and let $\{b\to Q_i\}_{i\in I}$ be a birhizomic cover of $b$.  Let $F$ be any presheaf of abelian groups on $B_P^{\rm op}$, and put $H:=\Lan F$.  Then the canonical map
\[
H(b)\longrightarrow
\ker\left(
\prod_i H(Q_i)
\rightrightarrows
\prod_{i,j} H(Q_i\oplus_b Q_j)
\right)
\]
is an isomorphism, where $Q_i\oplus_b Q_j$ denotes the pushout in $A_P$.
\end{lem}
\begin{proof}
Choose, by Lemma~\ref{lem:faithful-valuative}, an injective valuative map
\[
v:b\longrightarrow V.
\]
Because $\{b\to Q_i\}$ is a birhizomic cover, there is an index $i_0$ and a factorisation
\[
b\xrightarrow{f} Q_{i_0}\xrightarrow{g} V
\]
such that $f$ is birhizomic at $g$.

Since $v$ is injective, $f$ kills no element of $b$ except $0$.  Equivalently, the kernel of $P\to Q_{i_0}$ is exactly the face $F_x$.  Hence the comma category $B_P/Q_{i_0}$ has terminal object $b$, and therefore
\[
H(Q_{i_0})\cong H(b)=F(b).
\]

The birhizomic lifting property implies the following overlap statement: for every $i\in I$, the natural functor
\[
B_P/Q_i\longrightarrow B_P/(Q_i\oplus_b Q_{i_0})
\]
is cofinal.  Indeed, if a principal face became zero in the overlap but was not already zero on $Q_i$, the corresponding element would give an infinitesimal element relative to the valuative chart $f$, contradicting the unique lifting property in the definition of birhizomicity.  This is exactly the argument used in the construction of valuativisations via log blowups; see Lemma~2.7 and the surrounding discussion in~\href{https://pub.math.leidenuniv.nl/~holmesdst/notes_from_talks/valuativisations_talk.pdf}{pub.math.leidenuniv.nl}, and also~\href{https://arxiv.org/pdf/2411.13662}{arxiv.org} for the finite presentation formalism.

By cofinality, for every $i$ the restriction map
\[
H(Q_i)\longrightarrow H(Q_i\oplus_b Q_{i_0})
\]
is an isomorphism.

Now let $(s_i)_i$ be an element of the equaliser.  The component $s_{i_0}\in H(Q_{i_0})$ corresponds uniquely to an element $h\in H(b)$.  For each $i$, compatibility on the overlap $Q_i\oplus_b Q_{i_0}$ says that the image of $s_i$ equals the image of $s_{i_0}$.  Since the restriction map
\[
H(Q_i)\to H(Q_i\oplus_b Q_{i_0})
\]
is an isomorphism, it follows that $s_i$ is uniquely determined and equals the restriction of $h$.  Thus every matching family comes uniquely from $H(b)$.
\end{proof}

\begin{prop}\label{prop:unit-iso}
For every presheaf $F$ on $B_P^{\rm op}$, the unit
\[
F\longrightarrow \pi_*\pi^{-1}F
\]
is an isomorphism.
\end{prop}
\begin{proof}
Let $b=P_x\in B_P$ and put $H=\Lan F$.  First observe that $H(b)=F(b)$: the category $B_P/b$ has terminal object $b$.

Sheafification is computed objectwise by
\[
(a_CH)(b)=\varinjlim_{S} \Hom_{\PSh(C)}(S,H),
\]
where $S$ runs over covering sieves of $b$.  By definition of the birhizomic topology, every covering sieve contains a sieve generated by a birhizomic cover family.  Lemma~\ref{lem:contract} shows that for such a sieve $S_0$ the natural map
\[
H(b)\longrightarrow \Hom(S_0,H)
\]
is an isomorphism.  Passing to the filtered colimit over all covering sieves gives
\[
(a_CH)(b)\cong H(b)=F(b).
\]
Thus
\[
(\pi_*\pi^{-1}F)(b)=(a_C\Lan F)(b)\cong F(b),
\]
naturally in $b$ and $F$.
\end{proof}

\section{Derived conclusion}\label{sec:derived}

We now prove Theorem~\ref{thm:main}.

Let $F$ be a presheaf of abelian groups on $D=B_P^{\rm op}$.  Choose an injective resolution
\[
F\longrightarrow I^\bullet
\]
in $\PSh(D)$.  Since $\pi^{-1}$ is exact (Corollary~\ref{cor:pi-inv-exact}) and preserves injectives (Lemma~\ref{lem:preserves-injectives}), the complex
\[
\pi^{-1}I^\bullet
\]
is an injective resolution of $\pi^{-1}F$ in $\Sh(C)$.  Therefore
\[
R\pi_*\pi^{-1}F
\simeq
\pi_*\pi^{-1}I^\bullet.
\]
By Proposition~\ref{prop:unit-iso}, the unit is an isomorphism on every injective $I^n$, hence
\[
\pi_*\pi^{-1}I^\bullet \cong I^\bullet.
\]
Composing with the chosen quasi-isomorphism $F\to I^\bullet$ gives a quasi-isomorphism
\[
F\longrightarrow R\pi_*\pi^{-1}F.
\]
By functoriality of the derived unit, this is precisely the morphism in the statement of the theorem.

This completes the proof.
