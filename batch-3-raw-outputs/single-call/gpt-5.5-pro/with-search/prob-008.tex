\title{Global Hypoellipticity of a Vector Field Implies Zero Entropy}
\author{}
\date{}
\maketitle

\begin{abstract}
We prove that a smooth vector field \(X\) on a compact connected smooth
manifold whose Lie derivative is globally hypoelliptic on distributions
generates a flow of zero topological entropy.  The proof uses two standard
results: the Greenfield--Wallach--Chen--Chi--Forni theorem that globally
hypoelliptic vector fields are cohomology-free, and Katok's entropy obstruction
to finite-dimensional stable smooth cohomology.
\end{abstract}

\section{Preliminaries}

Let \(M\) be a compact connected smooth manifold and let \(X\in \mathfrak X^\infty(M)\).
Write \(g_t\) for the flow of \(X\), which is complete because \(M\) is compact.
Let \(\D'(M)\) denote the space of distributions on \(M\).  We use the convention
that \(Xf=\Lie_X f\) for \(f\in C^\infty(M)\), and the same symbol denotes the
transpose action on distributions.

\begin{definition}
The vector field \(X\) is called \emph{globally hypoelliptic} if
\[
        u\in \D'(M),\quad Xu\in C^\infty(M)
        \quad\Longrightarrow\quad
        u\in C^\infty(M).
\]
It is called \emph{\(C^\infty\)-cohomology-free} if for every
\(f\in C^\infty(M)\) there exist \(u\in C^\infty(M)\) and a constant \(c\in\C\)
such that
\[
        Xu=f-c .
\]
\end{definition}

We shall use the following two standard theorems.

\begin{theorem}[Greenfield--Wallach; Chen--Chi; Forni]
\label{thm:GH_CF}
Let \(M\) be a compact connected smooth manifold and let \(X\in\mathfrak X^\infty(M)\).
If \(X\) is globally hypoelliptic, then \(X\) is \(C^\infty\)-cohomology-free.
Moreover, the flow of \(X\) preserves a smooth probability density \(\mu\), the
space of \(X\)-invariant distributions is one-dimensional, namely
\[
        \ker(X:\D'(M)\to\D'(M))=\C\mu,
\]
and \(g_t\) is uniquely ergodic.
\end{theorem}

\begin{proof}[Reference]
This is the equivalence between global hypoellipticity and
cohomology-freeness proved in Greenfield--Wallach \cite{GW73}, refined by
Chen--Chi \cite{CC00}, and stated in the above form by Forni
\cite[Theorem 3.10]{Forni08}.  Forni's formulation does not require orientability
if smooth densities are used instead of volume forms.
\end{proof}

The second input is Katok's entropy obstruction to finite-dimensional stable
smooth cohomology.

\begin{theorem}[Katok's entropy obstruction]
\label{thm:Katok}
Let \(X\in\mathfrak X^\infty(M)\) generate a smooth flow \(g_t\) on a compact
manifold.  Suppose the range
\[
        X(C^\infty(M))\subset C^\infty(M)
\]
is closed and has finite codimension.  Then
\[
        \htop(g_t)=0 .
\]
Equivalently, if \(\htop(g_t)>0\), then the smooth cohomology
\[
        C^\infty(M)/X(C^\infty(M))
\]
is infinite-dimensional or the range \(X(C^\infty(M))\) is not closed.
\end{theorem}

\begin{proof}[Reference]
This is a standard consequence of Katok's combinatorial construction of smooth
cocycles over positive-entropy systems.  In the form used here, it appears in
Katok--Robinson \cite[Section 6]{KR01} and Katok \cite[Chapter 6]{Katok03}:
positive topological entropy produces arbitrarily many linearly independent
smooth cocycles modulo smooth coboundaries.  Thus a \(C^\infty\)-stable flow
with finite-dimensional smooth cohomology must have zero topological entropy.
\end{proof}

\section{Proof of the theorem}

\begin{theorem}
Let \(M\) be a compact connected smooth manifold and let \(X\in\mathfrak X^\infty(M)\).
Assume that whenever \(f\in C^\infty(M)\), \(u\in\D'(M)\), and
\[
        Xu=f
\]
in the sense of distributions, then \(u\in C^\infty(M)\).  Then the flow
\(g_t\) generated by \(X\) has zero topological entropy.
\end{theorem}

\begin{proof}
The stated hypothesis is precisely global hypoellipticity of the Lie derivative
\(X=\Lie_X\) on distributions.  By Theorem \ref{thm:GH_CF}, \(X\) is
\(C^\infty\)-cohomology-free.  Hence for every \(f\in C^\infty(M)\) there are
\(u\in C^\infty(M)\) and \(c\in\C\) such that
\[
        f=Xu+c .
\]
Therefore
\[
        C^\infty(M)=X(C^\infty(M))+\C\cdot 1 .
\]

Let \(\mu\) be the smooth invariant probability density supplied by
Theorem \ref{thm:GH_CF}.  Since \(X\mu=0\), we have
\[
        \int_M Xu\,d\mu=0
        \qquad\text{for every }u\in C^\infty(M).
\]
Consequently,
\[
        X(C^\infty(M))\subset
        \left\{f\in C^\infty(M):\int_M f\,d\mu=0\right\}.
\]
The cohomology-free property gives the reverse inclusion: if
\(\int_M f\,d\mu=0\), then in the decomposition \(f=Xu+c\) we must have
\(c=0\).  Thus
\[
        X(C^\infty(M))
        =
        \left\{f\in C^\infty(M):\int_M f\,d\mu=0\right\}.
\]
The right-hand side is a closed hyperplane in the Fr\'echet space
\(C^\infty(M)\).  Hence \(X(C^\infty(M))\) is closed and has codimension one.

Katok's entropy obstruction, Theorem \ref{thm:Katok}, now applies and yields
\[
        \htop(g_t)=0.
\]
Equivalently, \(\htop(g_1)=0\), and by the usual scaling law for flows,
\(\htop(g_t)=|t|\,\htop(g_1)=0\) for every \(t\in\R\).
\end{proof}

\begin{remark}
The proof uses substantially more than unique ergodicity.  Unique ergodicity
alone does not force zero topological entropy in general topological dynamics.
The essential point is the finite-dimensional, closed \(C^\infty\)-cohomology
forced by global hypoellipticity.
\end{remark}
