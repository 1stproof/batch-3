\title{Globally Hypoelliptic Vector Fields Have Zero Topological Entropy}
\date{}
\maketitle

\begin{theorem}\label{thm:main}
Let $M$ be a compact connected smooth manifold, let $X$ be a smooth vector field on $M$, and let $(g_t)_{t\in\mathbb R}$ be the flow generated by $X$. Assume that whenever $u\in\mathcal D'(M)$ and $f\in C^\infty(M)$ satisfy
\[
\mathcal L_X u=f,
\]
then $u\in C^\infty(M)$. Then the flow $g_t$ has zero topological entropy.
\end{theorem}

\medskip

The hypothesis is exactly the statement that the first-order operator $\mathcal L_X$ is \emph{globally hypoelliptic}. The proof is a short combination of two known results: a theorem of Greenfield--Wallach, which shows that globally hypoelliptic vector fields are cohomology-free, and a theorem of A.~Rodr\'iguez Hertz and F.~Rodr\'iguez Hertz, which shows that cohomology-free vector fields have zero topological entropy.

\section*{1. Preliminaries}

Since $M$ is compact, the flow of $X$ is complete.

\begin{lemma}\label{lem:cohfree-equiv}
For a smooth vector field $X$ on a compact connected manifold $M$, the following are equivalent:
\begin{enumerate}
\item[(i)] The quotient
\[
C^\infty(M)\big/\mathcal L_X\bigl(C^\infty(M)\bigr)
\]
is one-dimensional.
\item[(ii)] For every $f\in C^\infty(M)$ there exist $u\in C^\infty(M)$ and $c\in\mathbb R$ such that
\[
\mathcal L_X u=f-c.
\]
\end{enumerate}
\end{lemma}

\begin{proof}
Assume (ii). Then every class in the quotient is represented by a constant, so the quotient has dimension at most $1$.

It remains to see that the quotient is nonzero. Equivalently, we claim that
\[
1\notin \mathcal L_X\bigl(C^\infty(M)\bigr).
\]
Indeed, if $\mathcal L_Xu=1$ for some $u\in C^\infty(M)$, then along every orbit
\[
\frac{d}{dt}\,u(g_t(x)) = (\mathcal L_Xu)(g_t(x)) = 1,
\]
hence
\[
u(g_t(x)) = u(x)+t \qquad \forall\, t\in\mathbb R.
\]
This is impossible because $u$ is bounded on compact $M$. Thus the class of $1$ is nonzero, and the quotient is exactly one-dimensional.

Conversely, assume (i). Since the class of $1$ is nonzero by the argument above, it spans the one-dimensional quotient. Hence for every $f\in C^\infty(M)$ there exists $c\in\mathbb R$ such that
\[
[f]=c[1]
\]
in the quotient, i.e.
\[
f-c \in \mathcal L_X\bigl(C^\infty(M)\bigr).
\]
Thus there is $u\in C^\infty(M)$ with $\mathcal L_Xu=f-c$.
\end{proof}

\begin{definition}
A smooth vector field $X$ on a compact connected manifold is called \emph{cohomology-free} if it satisfies the equivalent conditions of Lemma~\ref{lem:cohfree-equiv}.
\end{definition}

\begin{remark}\label{rem:entropy-normalization}
For a continuous flow on a compact metric space, the topological entropy scales linearly with time:
\[
h_{\mathrm{top}}(g_t)=|t|\,h_{\mathrm{top}}(g_1)
\qquad (t\in\mathbb R),
\]
see Bowen--Walters~\cite{BW}. In particular, the statement that the flow has zero topological entropy is independent of the standard normalization.
\end{remark}

\begin{theorem}[Greenfield--Wallach {\cite{GW}}, see also {\cite[\S 2]{Forni}}]
\label{thm:GW}
Let $X$ be a smooth vector field on a compact connected manifold $M$. If $\mathcal L_X$ is globally hypoelliptic, then
\[
\dim\Bigl(C^\infty(M)\big/\mathcal L_X\bigl(C^\infty(M)\bigr)\Bigr)=1.
\]
Equivalently, $X$ is cohomology-free.
\end{theorem}

\begin{theorem}[A.~Rodr\'iguez Hertz--F.~Rodr\'iguez Hertz {\cite{RRH}}, see also {\cite[\S 2]{Forni}}]
\label{thm:RRH}
If $X$ is a cohomology-free smooth vector field on a compact manifold, then the flow generated by $X$ has zero topological entropy.
\end{theorem}

\section*{2. Proof of Theorem~\ref{thm:main}}

\begin{proof}[Proof of Theorem~\ref{thm:main}]
By hypothesis, $\mathcal L_X$ is globally hypoelliptic. Therefore Theorem~\ref{thm:GW} applies, and $X$ is cohomology-free. Then Theorem~\ref{thm:RRH} implies that the flow $(g_t)_{t\in\mathbb R}$ generated by $X$ has zero topological entropy.

By Remark~\ref{rem:entropy-normalization}, this is equivalent to the vanishing of the entropy of any nonzero time-$t$ map. This proves the theorem.
\end{proof}
