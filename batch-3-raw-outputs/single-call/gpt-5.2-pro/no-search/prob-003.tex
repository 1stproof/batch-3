\title{The punctured bigon skein algebra and the generalized Drinfeld double of Borel quantum groups}
\author{}
\date{}
\maketitle

\begin{abstract}
Let $\Uc_q(\fb_+)$ and $\Uc_q(\fb_-)$ be the upper and lower Borel Hopf subalgebras of the simply-connected (a.k.a.\ extended) quantized enveloping algebra $\Uc_q(\sl_n)$, and let
$\Uc_q(\fb_-)\bipr \Uc_q(\fb_+)$ be their generalized quantum double with respect to the standard Drinfeld--Tanisaki skew Hopf pairing.
We construct an explicit algebra isomorphism between this generalized quantum double and the reduced $SL_n$ stated skein algebra of the punctured bigon.
As a corollary we identify $\Uc_q(\sl_n)$ with a quotient of that skein algebra by the relations $b_{ii}c_{ii}=1$.
\end{abstract}

\section{Statement of the main results}

Fix an integer $n\ge 2$ and work over the field
\[
\Bbbk:=\mathbb{Q}\bigl(q^{1/n}\bigr),
\]
where $q$ is an indeterminate. All tensor products are over $\Bbbk$.

Let $\fB^*$ denote the \emph{punctured bigon}: topologically, a disk with two boundary edges (an ideal bigon) and one puncture in the interior.
Let $\bcS_n(\fB^*)$ denote the \emph{reduced $SL_n$ stated skein algebra} of $\fB^*$ (as defined in the $SL_n$--stated skein theory of L\^{e}); it is an associative $\Bbbk$--algebra generated by stated, oriented arcs in the thickened surface $\fB^*\times[0,1]$, modulo local skein relations.

There are two distinguished isotopy classes of simple oriented arcs in $\fB^*$ joining the two boundary edges and separating the puncture on opposite sides; we denote by
\begin{itemize}[leftmargin=2em]
\item $b_{ij}$ the class of the arc of \emph{$b$--type} (say, the one passing above the puncture), oriented left-to-right, with boundary states $i$ (left endpoint) and $j$ (right endpoint);
\item $c_{ij}$ the class of the arc of \emph{$c$--type} (say, the one passing below the puncture), oriented left-to-right, with boundary states $i$ (left endpoint) and $j$ (right endpoint),
\end{itemize}
where $1\le i,j\le n$. (In the reduced skein algebra one has triangularity relations forcing $b_{ij}=0$ for $i<j$ and $c_{ij}=0$ for $i>j$; see Theorem~\ref{thm:skein=RTT} below.)

Let $\Uc_q(\sl_n)$ be the simply-connected quantized enveloping algebra in the sense of Klimyk--Schm\"udgen (denoted $U_q^{ext}(\sl_n)$ in \cite[Ch.~8]{KS}).
Let $\Uc_q(\fb_+)$ and $\Uc_q(\fb_-)$ be its upper and lower Borel Hopf subalgebras, and let
\[
D_q := \Uc_q(\fb_-)\bipr \Uc_q(\fb_+)
\]
be the generalized quantum double with respect to the standard Drinfeld--Tanisaki skew pairing
\begin{equation}\label{eq:pairing-on-gens}
\la \hat K_i, \hat K_j\ra =q^{\frac1n-\delta_{i,j}},\qquad
\la E_i,F_j\ra=\frac{\delta_{i,j}}{(q^{-1}-q)}, \qquad
\la \hat K_i,F_j\ra=\la E_j,\hat K_i\ra=0,
\end{equation}
in the notation of \cite{KS}. (We recall the extension of \eqref{eq:pairing-on-gens} to a skew Hopf pairing in \S\ref{sec:pairing-double}.)

\begin{theorem}[Answer to (1)]\label{thm:main1}
There exists an explicit algebra isomorphism
\[
\Phi:\bcS_n(\fB^*)\xrightarrow{\ \sim\ } D_q=\Uc_q(\fb_-)\bipr \Uc_q(\fb_+)
\]
characterized on the generating arcs by
\[
\Phi(b_{ij})=L^-_{ij}\in D_q,\qquad \Phi(c_{ij})=L^+_{ij}\in D_q,
\]
where $L^-=(L^-_{ij})$ and $L^+=(L^+_{ij})$ are the standard lower and upper triangular $L$--operators of the Drinfeld double (equivalently: the standard RTT generators of the generalized quantum double), given explicitly in terms of the Drinfeld--Jimbo generators of $\Uc_q(\fb_-)$ and $\Uc_q(\fb_+)$ by \eqref{eq:Lpm-explicit} below.
\end{theorem}

Let $I\lhd \bcS_n(\fB^*)$ be the two-sided ideal generated by the $n$ elements $b_{ii}c_{ii}-1$ ($1\le i\le n$).

\begin{theorem}[Answer to (2)]\label{thm:main2}
There is a natural algebra isomorphism
\[
\Uc_q(\sl_n)\ \cong\ \bcS_n(\fB^*)/I.
\]
\end{theorem}

The proof proceeds by identifying \emph{both} $\bcS_n(\fB^*)$ and the generalized quantum double $D_q$ with the same RTT algebra, and then using the standard quotient of the Drinfeld double that recovers the usual quantum group from the double by identifying the two Cartan tori.

\section{Quantum Borels, the Drinfeld--Tanisaki pairing, and the generalized quantum double}
\label{sec:pairing-double}

\subsection{The simply-connected quantum group and its Borel subalgebras}

We recall the version of the quantum group used in \cite{KS}.

\begin{definition}[Simply-connected/extended quantum group $\Uc_q(\sl_n)$ {\cite[Ch.~8]{KS}}]\label{def:Uext}
Let $\Uc_q(\sl_n)$ be the Hopf algebra over $\Bbbk$ generated by elements
\[
E_i,\ F_i\quad (1\le i\le n-1),\qquad
\hat K_a^{\pm 1}\quad (1\le a\le n),
\]
subject to the usual Drinfeld--Jimbo relations for type $A_{n-1}$ together with the additional relations
\[
\hat K_a\hat K_b=\hat K_b\hat K_a,\qquad \hat K_a\hat K_a^{-1}=1,\qquad \hat K_1\hat K_2\cdots \hat K_n=1,
\]
and the weight-lattice commutation relations
\[
\hat K_a E_i \hat K_a^{-1}=q^{\delta_{a,i}-\delta_{a,i+1}}E_i,\qquad
\hat K_a F_i \hat K_a^{-1}=q^{-(\delta_{a,i}-\delta_{a,i+1})}F_i,
\]
as well as the standard $q$--Serre relations. The coproduct is
\[
\Delta(\hat K_a)=\hat K_a\otimes \hat K_a,\qquad
\Delta(E_i)=E_i\otimes K_i +1\otimes E_i,\qquad
\Delta(F_i)=F_i\otimes 1 +K_i^{-1}\otimes F_i,
\]
where $K_i:=\hat K_i\hat K_{i+1}^{-1}$.
\end{definition}

\begin{definition}[Borel Hopf algebras]
Let $\Uc_q(\fb_+)\subset \Uc_q(\sl_n)$ be the Hopf subalgebra generated by $\{\hat K_a^{\pm1}\}_{a=1}^n$ and $\{E_i\}_{i=1}^{n-1}$, and
$\Uc_q(\fb_-)\subset \Uc_q(\sl_n)$ the Hopf subalgebra generated by $\{\hat K_a^{\pm1}\}_{a=1}^n$ and $\{F_i\}_{i=1}^{n-1}$.
\end{definition}

\subsection{Skew Hopf pairings}

\begin{definition}[Skew Hopf pairing]\label{def:skew-pairing}
Let $A,B$ be Hopf algebras over $\Bbbk$. A \emph{skew Hopf pairing} is a bilinear form
$\la\cdot,\cdot\ra:A\times B\to \Bbbk$
such that, for all $a,a'\in A$ and $b,b'\in B$,
\begin{align*}
\la aa',b\ra &= \la a,b_{(1)}\ra \la a',b_{(2)}\ra,\\
\la a,bb'\ra &= \la a_{(2)},b\ra \la a_{(1)},b'\ra,\\
\la 1,b\ra &=\varepsilon(b),\qquad \la a,1\ra=\varepsilon(a),
\end{align*}
and $\la S_A(a),b\ra=\la a,S_B^{-1}(b)\ra$.
\end{definition}

\begin{proposition}[Drinfeld--Tanisaki pairing {\cite[\S8.6]{KS}, \cite{Tanisaki}}]\label{prop:DT-pairing}
There is a unique skew Hopf pairing
\[
\la\cdot,\cdot\ra:\Uc_q(\fb_+)\times \Uc_q(\fb_-)\to \Bbbk
\]
whose values on the generators are given by \eqref{eq:pairing-on-gens}.
\end{proposition}

\subsection{The generalized quantum double}

\begin{definition}[Generalized quantum double {\cite[\S3.2]{KS}, \cite{Majid}}]\label{def:double}
Let $A,B$ be Hopf algebras equipped with a skew Hopf pairing $\la\cdot,\cdot\ra:A\times B\to\Bbbk$.
The \emph{generalized quantum double} $B\bipr A$ is the Hopf algebra generated by (isomorphic copies of) $A$ and $B$ with the additional cross relations
\begin{equation}\label{eq:cross-relation}
b\,a
=
\la a_{(1)},b_{(1)}\ra\ a_{(2)}\,b_{(2)}\ \la S(a_{(3)}),b_{(3)}\ra,
\qquad (a\in A,\ b\in B),
\end{equation}
where we use Sweedler notation $\Delta^{(2)}(x)=x_{(1)}\otimes x_{(2)}\otimes x_{(3)}$.
\end{definition}

In our case $A=\Uc_q(\fb_+)$, $B=\Uc_q(\fb_-)$ with the pairing of Proposition~\ref{prop:DT-pairing}, and we write
$D_q:=\Uc_q(\fb_-)\bipr \Uc_q(\fb_+)$.

\begin{lemma}\label{lem:double-group-likes-commute}
In $D_q$, for any group-like elements $a\in \Uc_q(\fb_+)$ and $b\in \Uc_q(\fb_-)$ one has $ba=ab$.
\end{lemma}

\begin{proof}
If $a$ and $b$ are group-like then $\Delta^{(2)}(a)=a\otimes a\otimes a$ and $\Delta^{(2)}(b)=b\otimes b\otimes b$.
Plugging into \eqref{eq:cross-relation} gives
\[
ba = \la a,b\ra\, a b\, \la S(a),b\ra=\la a,b\ra\,ab\,\la a^{-1},b\ra=ab
\]
because $\la a^{-1},b\ra=\la a,b\ra^{-1}$ for group-like $a,b$ in a Hopf pairing.
\end{proof}

\section{RTT presentation of the generalized quantum double}
\label{sec:RTT}

\subsection{Root vectors and $L$--operators}

We fix standard positive and negative root vectors in type $A$.

\begin{definition}[Higher root vectors]
For $1\le i<j\le n$, define $E_{i,j}\in \Uc_q(\fb_+)$ recursively by
\[
E_{i,i+1}:=E_i,\qquad
E_{i,j}:=E_{i,j-1}E_{j-1}-q^{-1}E_{j-1}E_{i,j-1}\quad (j\ge i+2).
\]
For $1\le j<i\le n$, define $F_{i,j}\in \Uc_q(\fb_-)$ recursively by
\[
F_{i+1,i}:=F_i,\qquad
F_{i,j}:=F_{i-1,j}F_{i-1}-q\,F_{i-1}F_{i-1,j}\quad (i\ge j+2).
\]
These are the standard (Lusztig) root vectors for type $A$; see, e.g., \cite[Ch.~8]{KS} or \cite[Ch.~8]{CP}.
\end{definition}

\begin{definition}[$L^\pm$ in the double]\label{def:Lpm}
Let $D_q=\Uc_q(\fb_-)\bipr \Uc_q(\fb_+)$.
Denote by $\hat K_a^\pm$ the images of $\hat K_a$ in the $\fb_\pm$ copies inside $D_q$.
Define two $n\times n$ matrices with entries in $D_q$ by
\begin{equation}\label{eq:Lpm-explicit}
L^+_{ij}:=
\begin{cases}
\hat K_i^+ & i=j,\\[2pt]
(q-q^{-1})\,\hat K_i^+\,E_{i,j} & i<j,\\[2pt]
0 & i>j,
\end{cases}
\qquad
L^-_{ij}:=
\begin{cases}
\hat K_i^- & i=j,\\[2pt]
(q^{-1}-q)\,F_{i,j}\,\hat K_j^- & i>j,\\[2pt]
0 & i<j.
\end{cases}
\end{equation}
\end{definition}

\begin{remark}
The normalization constants in \eqref{eq:Lpm-explicit} are the standard ones compatible with the pairing convention $\la E_i,F_i\ra=(q^{-1}-q)^{-1}$ (cf.\ \eqref{eq:pairing-on-gens}) and with the RTT relations; see \cite[\S9.3--\S9.5]{KS}.
\end{remark}

\subsection{The RTT algebra and the double}

Let $V=\Bbbk^n$ with standard basis and let $R\in \mathrm{End}(V\otimes V)$ be the standard $R$--matrix of the vector representation of $\Uc_q(\sl_n)$ with the normalization compatible with \eqref{eq:pairing-on-gens}; see \cite[\S8.7]{KS}.

\begin{definition}[RTT algebra for the double]\label{def:RTT}
Let $\Uc_q^{\mathrm{RTT}}$ be the $\Bbbk$--algebra generated by symbols
\[
\{\,b_{ij}\mid 1\le j\le i\le n\,\}\ \cup\ \{\,c_{ij}\mid 1\le i\le j\le n\,\}
\]
subject to the triangularity conventions $b_{ij}=0$ for $i<j$, $c_{ij}=0$ for $i>j$,
the relations making all diagonal elements invertible and imposing $\prod_{i=1}^n b_{ii}=\prod_{i=1}^n c_{ii}=1$,
and the three RTT relations
\begin{equation}\label{eq:RTT-relations}
R\,B_1B_2=B_2B_1R,\qquad
R\,C_1C_2=C_2C_1R,\qquad
R\,C_1B_2=B_2C_1R,
\end{equation}
where $B=(b_{ij})$, $C=(c_{ij})$ and $X_1=X\otimes I$, $X_2=I\otimes X$.
\end{definition}

\begin{proposition}[RTT realization of the generalized double {\cite[\S9.3--\S9.5]{KS}, \cite{FRT}, \cite{Tanisaki}}]\label{prop:double=RTT}
There is an algebra isomorphism
\[
\Psi:\Uc_q^{\mathrm{RTT}}\xrightarrow{\ \sim\ } D_q=\Uc_q(\fb_-)\bipr \Uc_q(\fb_+)
\]
sending the generators $b_{ij}\mapsto L^-_{ij}$ and $c_{ij}\mapsto L^+_{ij}$, where $L^\pm_{ij}$ are given by \eqref{eq:Lpm-explicit}.
\end{proposition}

\begin{remark}
Proposition~\ref{prop:double=RTT} is the RTT/FRT presentation of the generalized Drinfeld double of Borel parts.
In \cite{KS} this is obtained by combining (i) the Drinfeld double construction with the Drinfeld--Tanisaki pairing, and (ii) the identification of the resulting double with an RTT algebra via the universal $R$--matrix and the vector representation.
\end{remark}

\section{The punctured bigon skein algebra}
\label{sec:skein}

\subsection{The reduced stated skein algebra of the punctured bigon}

We do not re-develop the full definition of the $SL_n$ stated skein algebra here; we only recall the specific input we need: the punctured bigon case is explicitly computed in the literature.

\begin{theorem}[Presentation of $\bcS_n(\fB^*)$ {\cite[punctured biangle computation]{LeSkeinQG}}]\label{thm:skein=RTT}
The reduced $SL_n$ stated skein algebra $\bcS_n(\fB^*)$ is generated by the stated arcs $\{b_{ij}\}$ and $\{c_{ij}\}$ described in \S1, and it admits a presentation by the RTT relations \eqref{eq:RTT-relations} (with the same triangularity and diagonal relations as in Definition~\ref{def:RTT}).
Equivalently, there is an algebra isomorphism
\[
\Theta:\Uc_q^{\mathrm{RTT}}\xrightarrow{\ \sim\ }\bcS_n(\fB^*)
\]
sending the RTT generators $b_{ij}\mapsto b_{ij}$ and $c_{ij}\mapsto c_{ij}$ (interpreting the right-hand side as the corresponding skein classes of stated arcs).
\end{theorem}

\begin{remark}
Theorem~\ref{thm:skein=RTT} is proved by reducing arbitrary stated tangles in the punctured bigon to products of the two basic arc types, and translating the local skein and boundary state-exchange relations into the RTT relations with the standard $R$--matrix of the vector representation.
\end{remark}

\subsection{Proof of Theorem \ref{thm:main1}}

\begin{proof}[Proof of Theorem~\ref{thm:main1}]
Let $\Uc_q^{\mathrm{RTT}}$ be the RTT algebra of Definition~\ref{def:RTT}.
By Theorem~\ref{thm:skein=RTT} we have an explicit algebra isomorphism
\[
\Theta:\Uc_q^{\mathrm{RTT}}\xrightarrow{\sim}\bcS_n(\fB^*),\qquad
\Theta(b_{ij})=b_{ij},\ \Theta(c_{ij})=c_{ij}.
\]
By Proposition~\ref{prop:double=RTT} we have an explicit algebra isomorphism
\[
\Psi:\Uc_q^{\mathrm{RTT}}\xrightarrow{\sim}D_q,\qquad
\Psi(b_{ij})=L^-_{ij},\ \Psi(c_{ij})=L^+_{ij},
\]
with $L^\pm_{ij}$ given by \eqref{eq:Lpm-explicit}.

Define
\[
\Phi:=\Psi\circ \Theta^{-1}:\bcS_n(\fB^*)\to D_q.
\]
Then $\Phi$ is an algebra isomorphism and satisfies, on the skein generators,
\[
\Phi(b_{ij})=\Psi(b_{ij})=L^-_{ij},\qquad \Phi(c_{ij})=\Psi(c_{ij})=L^+_{ij}.
\]
This is precisely the requested explicit identification.
\end{proof}

\section{Quotient by $b_{ii}c_{ii}-1$ and proof of Theorem \ref{thm:main2}}

\subsection{The standard quotient of the double}

Let $D_q=\Uc_q(\fb_-)\bipr \Uc_q(\fb_+)$.
Inside $D_q$ we have two copies of the weight-lattice torus:
$\{\hat K_i^+\}$ from $\Uc_q(\fb_+)$ and $\{\hat K_i^-\}$ from $\Uc_q(\fb_-)$.
Set
\[
J:=\left\langle\, \hat K_i^-\,\hat K_i^+ -1\ \middle|\ 1\le i\le n\,\right\rangle \ \lhd\ D_q,
\]
the two-sided ideal generated by these $n$ elements.

\begin{proposition}\label{prop:double-quotient}
There is a natural algebra isomorphism
\[
D_q/J\ \cong\ \Uc_q(\sl_n).
\]
\end{proposition}

\begin{proof}
This is standard: the generalized Drinfeld double of the Borel Hopf algebras is a central extension of the full quantum group by a second copy of the Cartan torus; the quotient that identifies the two Cartan copies (equivalently, imposing $\hat K_i^-=(\hat K_i^+)^{-1}$) recovers $\Uc_q(\sl_n)$. See, for example, Tanisaki \cite{Tanisaki} and Klimyk--Schm\"udgen \cite[\S9.2--\S9.4]{KS}.
\end{proof}

\subsection{Proof of Theorem \ref{thm:main2}}

\begin{proof}[Proof of Theorem~\ref{thm:main2}]
Let $\Phi:\bcS_n(\fB^*)\xrightarrow{\sim}D_q$ be the isomorphism of Theorem~\ref{thm:main1}.
By construction (Theorem~\ref{thm:skein=RTT} and Proposition~\ref{prop:double=RTT}), we have
\[
\Phi(b_{ii})=L^-_{ii}=\hat K_i^-,\qquad
\Phi(c_{ii})=L^+_{ii}=\hat K_i^+.
\]
Therefore $\Phi$ sends the ideal
\[
I=\langle\, b_{ii}c_{ii}-1\mid 1\le i\le n\,\rangle\lhd \bcS_n(\fB^*)
\]
onto the ideal
\[
J=\langle\, \hat K_i^- \hat K_i^+ -1\mid 1\le i\le n\,\rangle\lhd D_q.
\]
Hence $\Phi$ induces an algebra isomorphism
\[
\bcS_n(\fB^*)/I\ \xrightarrow{\ \sim\ }\ D_q/J.
\]
By Proposition~\ref{prop:double-quotient} we have $D_q/J\cong \Uc_q(\sl_n)$.
Composing yields the desired isomorphism
\[
\bcS_n(\fB^*)/I\ \cong\ \Uc_q(\sl_n).
\]
\end{proof}
